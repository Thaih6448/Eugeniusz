cmake_minimum_required(VERSION 3.20)
project(Eugeniusz VERSION 0.1.0 LANGUAGES C CXX)
include(GNUInstallDirs)
include(CTest)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/lib")
if(APPLE)
  set(CMAKE_INSTALL_RPATH "@loader_path")
elseif(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN")
endif()
option(BUILD_SHARED_LIBS "Build shared libraries" ON)
option(EUGENIUSZ_STATIC_RUNTIME "Embed the MSVC runtime in native libraries (requires validation with the selected backend)" OFF)
if(MSVC AND EUGENIUSZ_STATIC_RUNTIME)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
option(EUGENIUSZ_LLAMA "Build the optional local GGUF inference library" OFF)
option(EUGENIUSZ_EXAMPLES "Build examples" ON)
set(EUGENIUSZ_ACCELERATOR "CPU" CACHE STRING "CPU, CUDA, VULKAN, or METAL")
set_property(CACHE EUGENIUSZ_ACCELERATOR PROPERTY STRINGS CPU CUDA VULKAN METAL)
set(EUGENIUSZ_LLAMA_SOURCE "" CACHE PATH "Existing llama.cpp b6500 source; empty downloads the pinned revision")
find_package(Threads REQUIRED)
add_library(eugeniusz src/core.cpp)
add_library(Eugeniusz::eugeniusz ALIAS eugeniusz)
target_compile_features(eugeniusz PUBLIC cxx_std_17)
target_include_directories(eugeniusz PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_link_libraries(eugeniusz PRIVATE Threads::Threads)
target_compile_definitions(eugeniusz PRIVATE EG_BUILD EG_VERSION="${PROJECT_VERSION}")
if(NOT BUILD_SHARED_LIBS)
  target_compile_definitions(eugeniusz PUBLIC EG_STATIC)
endif()
set_target_properties(eugeniusz PROPERTIES VERSION ${PROJECT_VERSION} SOVERSION 0 CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(eugeniusz PRIVATE /W4 /permissive- /utf-8)
else()
  target_compile_options(eugeniusz PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(EUGENIUSZ_LLAMA)
  if(NOT EUGENIUSZ_ACCELERATOR MATCHES "^(CPU|CUDA|VULKAN|METAL)$")
    message(FATAL_ERROR "Unknown EUGENIUSZ_ACCELERATOR")
  endif()
  if(EUGENIUSZ_ACCELERATOR STREQUAL "METAL" AND NOT APPLE)
    message(FATAL_ERROR "Metal requires macOS")
  endif()
  set(GGML_NATIVE OFF CACHE BOOL "Portable CPU build" FORCE)
  set(GGML_CUDA OFF CACHE BOOL "" FORCE)
  set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
  set(GGML_METAL OFF CACHE BOOL "" FORCE)
  set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
  set(GGML_BACKEND_DL OFF CACHE BOOL "" FORCE)
  if(NOT EUGENIUSZ_ACCELERATOR STREQUAL "CPU")
    set(GGML_${EUGENIUSZ_ACCELERATOR} ON CACHE BOOL "" FORCE)
  endif()
  set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_SERVER OFF CACHE BOOL "" FORCE)
  set(LLAMA_CURL OFF CACHE BOOL "" FORCE)
  set(LLAMA_BUILD_COMMON OFF CACHE BOOL "" FORCE)
  function(eugeniusz_add_llama)
    # The pinned b6500 source uses u8 literals as char strings. Keep its
    # language standard local so consumers can build our targets with C++20.
    set(CMAKE_CXX_STANDARD 17)
    if(EUGENIUSZ_LLAMA_SOURCE)
      add_subdirectory("${EUGENIUSZ_LLAMA_SOURCE}" "${CMAKE_BINARY_DIR}/llama")
    else()
      include(FetchContent)
      FetchContent_Declare(llama URL https://codeload.github.com/ggml-org/llama.cpp/zip/a7a98e0fffed794396b3fbad4dcdbbc184963645 URL_HASH SHA256=a79b06ce6c36cbe36f162ace8ca0c043e707cc53390aae3a32003af163768768)
      FetchContent_MakeAvailable(llama)
    endif()
  endfunction()
  eugeniusz_add_llama()
  add_library(eugeniusz_llama src/llama_backend.cpp)
  add_library(Eugeniusz::llama ALIAS eugeniusz_llama)
  target_compile_features(eugeniusz_llama PRIVATE cxx_std_17)
  target_compile_definitions(eugeniusz_llama PRIVATE EG_LLAMA_BUILD)
  if(NOT BUILD_SHARED_LIBS)
    target_compile_definitions(eugeniusz_llama PUBLIC EG_STATIC)
  endif()
  target_link_libraries(eugeniusz_llama PUBLIC eugeniusz PRIVATE llama ${CMAKE_DL_LIBS})
  set_target_properties(eugeniusz_llama PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES VERSION ${PROJECT_VERSION} SOVERSION 0)
  install(TARGETS eugeniusz_llama RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
endif()
if(EUGENIUSZ_EXAMPLES)
  add_executable(eugeniusz_math examples/calibration.cpp)
  target_link_libraries(eugeniusz_math PRIVATE eugeniusz)
  if(EUGENIUSZ_LLAMA)
    add_executable(eugeniusz_decide examples/decide.cpp)
    target_link_libraries(eugeniusz_decide PRIVATE eugeniusz_llama)
  endif()
endif()
if(BUILD_TESTING)
  add_executable(eugeniusz_tests tests/core_test.cpp)
  target_link_libraries(eugeniusz_tests PRIVATE eugeniusz Threads::Threads)
  add_test(NAME core COMMAND eugeniusz_tests)
  add_executable(eugeniusz_c_test tests/c_api_test.c)
  target_link_libraries(eugeniusz_c_test PRIVATE eugeniusz)
  add_test(NAME c_api COMMAND eugeniusz_c_test)
endif()
install(TARGETS eugeniusz EXPORT EugeniuszTargets RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR} LIBRARY DESTINATION ${CMAKE_INSTALL_LIBDIR} ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
include(CMakePackageConfigHelpers)
configure_package_config_file(cmake/EugeniuszConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/EugeniuszConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Eugeniusz)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/EugeniuszConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(EXPORT EugeniuszTargets NAMESPACE Eugeniusz:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Eugeniusz)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/EugeniuszConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/EugeniuszConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/Eugeniusz)
install(FILES LICENSE THIRD_PARTY_NOTICES.md DESTINATION ${CMAKE_INSTALL_DATADIR}/eugeniusz)
install(DIRECTORY third_party/licenses/ DESTINATION ${CMAKE_INSTALL_DATADIR}/eugeniusz/licenses)
